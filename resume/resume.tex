% === Building blocks ============================================
% A key: dates and other metadata, in small mono.
\newcommand{\cvkey}[1]{{\footnotesize\ttfamily #1}}

% Glue that outranks the fil glue \raggedright puts at the end of
% each line, so keys sit flush right. A plain \hfill would share
% the leftover space with \rightskip and stop short of the margin.
\newcommand{\cvfill}{\hspace{0pt plus 1filll}}

% \cvemployer{employer}{key}: employer a size up, key flush right.
\newcommand{\cvemployer}[2]{\par{\large #1}\cvfill\cvkey{#2}\par}

% \cvcontext{text}: one short paragraph of context, set small.
\newcommand{\cvcontext}[1]{{\small #1\par}}

% \cvrole{title}{dates}: bold title, dates flush right.
\newcommand{\cvrole}[2]{\par\textbf{#1}\cvfill\cvkey{#2}\par}

% \cvproject{name}{what it is}{stack}{key}: stack in italics,
% key (status, repo link) flush right.
\newcommand{\cvproject}[4]{%
  \par\textbf{#1}: #2\quad\textit{#3}\cvfill\cvkey{#4}\par}

% \cvskill{label}{items}: label in a fixed-width key column.
\newlength{\cvkeywidth}
\newcommand{\cvskill}[2]{%
  \par\hangindent=\cvkeywidth\hangafter=1
  \makebox[\cvkeywidth][l]{\cvkey{#1}}#2\par}

\begin{itemize}
  \item Built the location service: drivers stream GPS pings over WebSocket into Postgres,

        TTL keys and a background sweeper evict stale drivers.
